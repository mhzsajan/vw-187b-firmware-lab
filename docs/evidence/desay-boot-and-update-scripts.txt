########## ./etc/func_env.sh  (3098 bytes)
#!/bin/sh
svp_env_cgroup()
{
	if [ "$SVP_ENABLE_CGROUP" != "true" ]; then 
		return 0
	fi

	if [ ! -x "/usr/bin/cgexec" ]; then
		export SVP_ENABLE_CGROUP="false"
		return 0
	fi

	echo "[SVP] Create cgroups ..."

	mkdir /run/cgroups/cpu -p
	mount -t cgroup -o cpu cpu /run/cgroups/cpu
	mkdir /run/cgroups/cpu/svp
	mkdir /run/cgroups/cpu/app
	mkdir /run/cgroups/cpu/specific
    mkdir /run/cgroups/cpu/link
	cgset -r cpu.shares=341 link

	return 0
}

svp_env_procguard()
{
	if [ "$SVP_ENABLE_PROCGUARD" != "true" ]; then 
		return 0
	fi

	if [ ! -x "${SVP_ROOT_PATH}/bin/procguard" ]; then
		export SVP_ENABLE_PROCGUARD="false"
	fi

	return 0
}

svp_env_normal()
{
	echo "[SVP] Setup ENV for Normal Boot-Up ..."
	# Mount "data" Partition
	mount ${SVP_DATA_PARTITION} ${SVP_USER_ROOT_PATH}

	if [ $? -ne 0 ]; then
		if [ -s "${SVP_USER_ROOT_PATH}/lost+found" ]; then
			echo "[SVP] mount successful..."
		else
			echo "[SVP] mount fail, formatting..."
			mkfs.ext4 ${SVP_DATA_PARTITION}
			mount ${SVP_DATA_PARTITION} ${SVP_USER_ROOT_PATH}
		fi
	else
		echo "[SVP] mount return 0..."
	fi

	if [ ! -d "${SVP_USER_APP_INSTALL_PATH}" ]; then
		mkdir -p ${SVP_USER_APP_INSTALL_PATH}
	fi

	if [ ! -d "${SVP_USER_DATA_PATH}/appinfo" ]; then
		mkdir -p ${SVP_USER_DATA_PATH}/appinfo
	fi

    if [ ! -d "${SVP_USER_DATA_PATH}/kanzi" ]; then
        mkdir -p ${SVP_USER_DATA_PATH}/kanzi
    fi

    if [ ! -d "${SVP_USER_ETC_PATH}" ]; then
        mkdir -p ${SVP_USER_ETC_PATH}
    fi

	# Deploy "ipod_config"
	if [ ! -s "${SVP_USER_ROOT_PATH}/ipod_config.xml" ]; then
		cp -a /svp/bin/ipod_config.xml ${SVP_USER_ROOT_PATH}
	fi

	if [ ! -s "${SVP_USER_DATA_PATH}/HWR_SYS_SIN_UNI" ]; then
		cp -a /svp/share/HWR_SYS_SIN_UNI ${SVP_USER_DATA_PATH}
	fi
	
	if [ -e "${SVP_USER_ROOT_PATH}/debug.flag" ]; then
		debug_flag=`cat "${SVP_USER_ROOT_PATH}/debug.flag"`
		case "$debug_flag" in
			null|none)
				export SVP_DEBUG_OUTPUT="none"
				;;
			kmsg|syslog)
				export SVP_DEBUG_OUTPUT="kmsg"
				;;
			file)
				export SVP_DEBUG_OUTPUT="file"
				;;
			*)
				export SVP_DEBUG_OUTPUT="console"
				;;
		esac
	fi

	svp_env_cgroup

	svp_env_procguard

	return 0
}

svp_env_update()
{
	echo "[SVP] Setup ENV for SW Update ..."
	mount ${SVP_STORAGE_PATH} -o rw,remount
	mkdir -p ${SVP_USER_ROOT_PATH}

	if [ $SVP_UPDATE_SW_CLEAN = "YES" ]; then
		#升级过程中需要格式化数据保存区
		mkfs.ext4 ${SVP_DATA_PARTITION}
		mount ${SVP_DATA_PARTITION} ${SVP_USER_ROOT_PATH}
	else
		mount ${SVP_DATA_PARTITION} ${SVP_USER_ROOT_PATH}
	    #删除trace开关文件，升级后默认没有trace输出
	    rm -fr /storage/data/debug.flag	
	fi

	mkdir -p ${SVP_SYS_APP_INSTALL_PATH}
	#mkdir -p ${SVP_SYS_DATA_PATH}
	mkdir -p ${SVP_SYS_DATA_PATH}/appinfo

	mkdir -p ${SVP_USER_APP_INSTALL_PATH}
	#mkdir -p ${SVP_USER_DATA_PATH}
	mkdir -p ${SVP_USER_DATA_PATH}/appinfo
	mkdir -p ${SVP_USER_ETC_PATH}

	svp_env_cgroup

	svp_env_procguard

	return 0
}

svp_env_update_rmt()
{
	echo "[SVP] Setup ENV for SW Update (RMT Only) ..."
	mount ${SVP_STORAGE_PATH} -o rw,remount

	svp_env_cgroup

	svp_env_procguard

	return 0
}
